### 步骤 3：实现 Benchmark（Buy & Hold）计算

**文件**：`services/control-api/backtest_engine.py`

在 `_build_metrics` 函数之后新增：

```python
def _compute_buy_hold_return(candles: List[CandlePoint]) -> float:
    """计算 Buy & Hold 总收益百分比。

    用第一根 K 线的 open 和最后一根 K 线的 close 计算。
    返回百分比，如 32.1 表示 +32.1%。
    """
    if len(candles) < 2:
        return 0.0
    entry_price = candles[0].open
    exit_price = candles[-1].close
    if entry_price <= 0:
        return 0.0
    return ((exit_price / entry_price) - 1.0) * 100.0
```

**集成位置**：在 `_run_trend_follow`、`_run_mean_reversion`、`_run_breakout` 三个策略函数内，计算完 metrics 之后、return 之前，将 benchmark 信息附加到 metrics 上。

### 步骤 4：实现 Walk-Forward Validation 引擎

**文件**：`services/control-api/backtest_engine.py`

在文件末尾（`run_local_backtest` 之后）新增以下函数：

```python
# ---------------------------------------------------------------------------
# Walk-Forward Validation
# ---------------------------------------------------------------------------

WALK_FORWARD_MIN_CANDLES = 200
# 至少需要 200 根 K 线才启用 walk-forward
# 200 根按 1h 约 8 天，按 4h 约 33 天，按 1d 约 200 天

WALK_FORWARD_MIN_WINDOW_CANDLES = 40
# 每个窗口（训练段或测试段）至少 40 根 K 线

WALK_FORWARD_TRAIN_RATIO = 0.7
# 每个窗口中训练段占比 70%，测试段占比 30%

WALK_FORWARD_MAX_WINDOWS = 8
# 最多划分 8 个窗口，避免计算量过大

WALK_FORWARD_MIN_WINDOWS = 3
# 至少需要 3 个窗口才有统计意义


@dataclass
class _WalkForwardWindowInput:
    """单个前推验证窗口的输入数据切分"""
    window_index: int
    train_candles: List[CandlePoint]
    test_candles: List[CandlePoint]


def _split_walk_forward_windows(
    candles: List[CandlePoint],
    train_ratio: float = WALK_FORWARD_TRAIN_RATIO,
    min_windows: int = WALK_FORWARD_MIN_WINDOWS,
    max_windows: int = WALK_FORWARD_MAX_WINDOWS,
    min_window_candles: int = WALK_FORWARD_MIN_WINDOW_CANDLES,
) -> List[_WalkForwardWindowInput]:
    """将 K 线序列切分为多个前推验证窗口。

    切分策略：anchored walk-forward（锚定前推）。
    - 窗口 0: candles[0 : split_0] 训练, candles[split_0 : end_0] 测试
    - 窗口 1: candles[0 : split_1] 训练, candles[split_1 : end_1] 测试
    - ...
    每个后续窗口的训练段都从头开始，包含前一个窗口的测试段。
    这样训练段逐渐变长，测试段保持固定大小，模拟"随时间积累经验"的真实场景。

    返回空列表表示数据量不足以做 walk-forward。
    """
    total = len(candles)
    if total < WALK_FORWARD_MIN_CANDLES:
        return []

    # 计算测试段大小：总量的 (1 - train_ratio) / window_count
    # 先确定窗口数
    min_test_size = min_window_candles
    min_train_size = min_window_candles

    # 可用的测试段总量 = total * (1 - train_ratio)
    available_test = int(total * (1.0 - train_ratio))
    window_count = max(min_windows, min(max_windows, available_test // min_test_size))

    # 如果算出来的窗口数不够最小值，降级
    if window_count < min_windows:
        return []

    test_size = available_test // window_count
    if test_size < min_test_size:
        return []

    windows: List[_WalkForwardWindowInput] = []
    # 初始训练段结束位置
    base_train_end = int(total * train_ratio) - (window_count - 1) * test_size

    if base_train_end < min_train_size:
        return []

    for i in range(window_count):
        train_end = base_train_end + i * test_size
        test_end = train_end + test_size
        if test_end > total:
            test_end = total
        if train_end >= total or test_end <= train_end:
            break
        train_slice = candles[:train_end]
        test_slice = candles[train_end:test_end]
        if len(train_slice) < min_train_size or len(test_slice) < min_test_size:
            continue
        windows.append(_WalkForwardWindowInput(
            window_index=i,
            train_candles=train_slice,
            test_candles=test_slice,
        ))

    if len(windows) < min_windows:
        return []

    return windows


def _run_strategy_on_candles(
    strategy: StrategySummary,
    candles: List[CandlePoint],
    timeframe: str,
) -> tuple[BacktestMetrics, bool]:
    """在给定 K 线上运行策略，返回 metrics 和 reference_only 标志。

    复用已有的三个策略运行函数。
    """
    params = _parameter_map(strategy)
    benchmark = _compute_buy_hold_return(candles)
    if strategy.id.startswith("trend-"):
        metrics, ref = _run_trend_follow(candles, params, timeframe)
    elif strategy.id.startswith("eth-revert") or "均值回归" in strategy.name:
        metrics, ref = _run_mean_reversion(candles, params, timeframe)
    else:
        metrics, ref = _run_breakout(candles, params, timeframe)
    metrics.benchmark_return = _format_signed_pct(benchmark)
    total_return = _parse_pct(metrics.annual_return)
    metrics.excess_return = _format_signed_pct(total_return - benchmark)
    return metrics, ref


def _parse_pct(formatted: str) -> float:
    """解析格式化的百分比字符串回 float，如 '+48.2%' → 48.2"""
    try:
        return float(formatted.replace("%", "").replace("+", "").strip())
    except (ValueError, AttributeError):
        return 0.0


def run_walk_forward(
    strategy: StrategySummary,
    candles: List[CandlePoint],
    timeframe: str,
) -> Optional[WalkForwardResult]:
    """对给定策略和 K 线数据执行 Walk-Forward 前推验证。

    如果数据量不足（< WALK_FORWARD_MIN_CANDLES），返回 None。
    """
    # 这里导入在函数内部，避免循环依赖——实际编码时根据模块结构调整
    from models import WalkForwardResult, WalkForwardWindowResult

    windows = _split_walk_forward_windows(candles)
    if not windows:
        return None

    bars_per_year = 365 * 24 / _timeframe_hours(timeframe)
    window_results: List[WalkForwardWindowResult] = []

    for window in windows:
        train_metrics, _ = _run_strategy_on_candles(strategy, window.train_candles, timeframe)
        test_metrics, _ = _run_strategy_on_candles(strategy, window.test_candles, timeframe)

        train_return = _parse_pct(train_metrics.pnl)  # 用 pnl 的百分比表示不准确，改用下面的方法
        # 需要从 equity 计算总收益百分比，但当前 _run_* 只返回 metrics，不返回 equity_curve
        # 所以我们直接解析 annual_return 作为近似
        # 注意：annual_return 是年化后的，不是总收益——需要新增一个 total_return 字段
        # 见下方"补充改动"

        train_total_return = _parse_pct(train_metrics.annual_return)
        test_total_return = _parse_pct(test_metrics.annual_return)
        train_sharpe = float(train_metrics.sharpe)
        test_sharpe = float(test_metrics.sharpe)
        train_mdd = _parse_pct(train_metrics.max_drawdown)
        test_mdd = _parse_pct(test_metrics.max_drawdown)
        test_benchmark = _compute_buy_hold_return(window.test_candles)

        train_start, train_end = _candle_time_bounds(window.train_candles)
        test_start, test_end = _candle_time_bounds(window.test_candles)

        window_results.append(WalkForwardWindowResult(
            window_index=window.window_index,
            train_start=train_start or "",
            train_end=train_end or "",
            test_start=test_start or "",
            test_end=test_end or "",
            train_candle_count=len(window.train_candles),
            test_candle_count=len(window.test_candles),
            train_return_pct=round(train_total_return, 2),
            test_return_pct=round(test_total_return, 2),
            train_sharpe=round(train_sharpe, 2),
            test_sharpe=round(test_sharpe, 2),
            train_max_drawdown_pct=round(train_mdd, 2),
            test_max_drawdown_pct=round(test_mdd, 2),
            train_trades=train_metrics.trades,
            test_trades=test_metrics.trades,
            test_benchmark_return_pct=round(test_benchmark, 2),
            degradation_pct=round(train_total_return - test_total_return, 2),
        ))

    if not window_results:
        return None

    avg_test_return = mean(w.test_return_pct for w in window_results)
    avg_test_sharpe = mean(w.test_sharpe for w in window_results)
    avg_test_mdd = mean(w.test_max_drawdown_pct for w in window_results)
    avg_degradation = mean(w.degradation_pct for w in window_results)
    worst_test_return = min(w.test_return_pct for w in window_results)
    positive_windows = sum(1 for w in window_results if w.test_return_pct > 0)
    consistency_score = round((positive_windows / len(window_results)) * 100.0, 1)

    if consistency_score >= 70 and avg_degradation < 15:
        stability_rating = "stable"
    elif consistency_score >= 50 and avg_degradation < 30:
        stability_rating = "moderate"
    else:
        stability_rating = "unstable"

    notes_parts = [
        f"共 {len(window_results)} 个前推窗口",
        f"测试段平均收益 {avg_test_return:+.1f}%",
        f"一致性 {consistency_score:.0f}%",
        f"稳定性 {stability_rating}",
    ]
    if avg_degradation > 10:
        notes_parts.append(f"平均退化 {avg_degradation:+.1f}%，训练段表现明显优于测试段，存在过拟合风险")
    if worst_test_return < -5:
        notes_parts.append(f"最差窗口测试收益 {worst_test_return:+.1f}%，需关注极端时段")

    return WalkForwardResult(
        window_count=len(window_results),
        windows=window_results,
        avg_test_return_pct=round(avg_test_return, 2),
        avg_test_sharpe=round(avg_test_sharpe, 2),
        avg_test_max_drawdown_pct=round(avg_test_mdd, 2),
        avg_degradation_pct=round(avg_degradation, 2),
        worst_test_return_pct=round(worst_test_return, 2),
        consistency_score=consistency_score,
        stability_rating=stability_rating,
        total_test_candles=sum(w.test_candle_count for w in window_results),
        total_train_candles=sum(w.train_candle_count for w in window_results),
        notes="；".join(notes_parts),
    )
```

```python
    metrics.benchmark_return = _format_signed_pct(benchmark_return_pct)
    total_return_pct = _parse_pct(metrics.total_return)
    metrics.excess_return = _format_signed_pct(total_return_pct - benchmark_return_pct)

    # --- Walk-Forward 前推验证（仅在非 reference_only 且数据充足时执行）---
    if not used_reference_path and used_candle_count >= WALK_FORWARD_MIN_CANDLES:
        metrics.walk_forward = run_walk_forward(strategy, effective_candles, timeframe)

    # --- 构建 notes ---
    notes = (
        f"由本地回测引擎基于 Bybit 历史 {timeframe} K 线生成，区间 {data_range}，"
        "当前包含简化手续费/滑点与策略规则近似，不再使用 mock 固定结果。"
    )
    if history_truncated:
        notes += f" 本次共取到 {retrieved_candle_count} 根 K 线，但回测内核当前仅使用最近 {used_candle_count} 根样本。"
    if used_reference_path:
        notes += " 本次区间未命中真实入场信号，收益/回撤基于基础价格路径生成的参考权益曲线，仅供研究参考。"
    if metrics.walk_forward is not None:
        notes += f" Walk-Forward 验证：{metrics.walk_forward.notes}。"

    return BacktestComputation(
        metrics=metrics,
        notes=notes,
        parameter_snapshot=params,
        symbol_scope=list(strategy.symbols),
        data_granularity="kline",
        reference_only=used_reference_path,
        sample_quality=derive_backtest_sample_quality(used_reference_path, metrics.trades),
        retrieved_candle_count=retrieved_candle_count,
        used_candle_count=used_candle_count,
        retrieved_range_start=retrieved_range_start,
        retrieved_range_end=retrieved_range_end,
        used_range_start=used_range_start,
        used_range_end=used_range_end,
        history_truncated=history_truncated,
    )
```

**不需要修改** `main.py:build_backtest_payload`，因为它已经在使用 `computation.metrics` 赋值给 BacktestRun，新字段会自动透传。

## 四、关键设计决策说明

### 4.1 为什么用 Anchored Walk-Forward 而不是 Rolling Walk-Forward

Rolling Walk-Forward 的训练窗口是固定大小的滑动窗口，而 Anchored 的训练窗口从数据起点开始逐渐变长。选择 Anchored 的原因：

1. 你的策略参数是固定的（不做重新优化），所以训练窗口的作用是"验证"而非"优化"
2. Anchored 更接近实际使用场景——你的策略累积了所有历史经验
3. 计算量更小——不需要对每个窗口都从头跑策略

### 4.2 为什么 walk_forward 放在 BacktestMetrics 里而不是 BacktestRun 里

1. `BacktestMetrics` 是回测引擎的直接输出，walk-forward 是对同一组数据的验证分析，逻辑上属于指标的一部分
2. `BacktestRun` 已经有 60+ 个字段了，不宜继续膨胀
3. `BacktestMetrics` 被嵌入到多个地方（`BacktestRun.metrics`、`StrategyActivityBacktestSummary` 等），walk-forward 结果可以自然透传

### 4.3 为什么 Benchmark 用 Buy & Hold 而不是市场指数

1. 你的策略是单币种的（每个策略对应一个 symbol），Buy & Hold 是最直接的基准
2. 不需要额外数据源
3. 加密市场没有公认的 benchmark index

### 4.4 向后兼容性

所有新增字段都有默认值：
- `benchmark_return = "+0.0%"`
- `excess_return = "+0.0%"`
- `total_return = "+0.0%"`
- `walk_forward = None`

已有的 seed 数据、历史 BacktestRun 记录在反序列化时不会报错。前端新增字段用 `?.` 和条件渲染保护。
